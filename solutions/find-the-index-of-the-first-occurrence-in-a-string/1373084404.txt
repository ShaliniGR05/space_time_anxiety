class Solution {
public:
    int strStr(string haystack, string needle) {
        int n = haystack.length();
        int n1 = needle.length();
        if(n1>n){
            return -1;
        }
        else{
            for(int i=0;i<=n-n1;i++){
                if(haystack.substr(i,n1) == needle){
                    return i;
                }
            }
        }
        return -1;
    }
};
